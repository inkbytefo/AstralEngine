# Open world (M9): a 4 x 4 km map generated by tools/make_test_world.py at build time (thousands of
# entity files stay out of the repository), cooked into streaming cells by astral-cook.
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    set(open_world_dir ${CMAKE_CURRENT_BINARY_DIR}/open_world_data)
    add_custom_command(
        OUTPUT ${open_world_dir}/source/maps/test.world
        COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/tools/make_test_world.py ${open_world_dir}/source
        DEPENDS ${CMAKE_SOURCE_DIR}/tools/make_test_world.py
        COMMENT "Generating the open world test map"
        VERBATIM)
    add_custom_command(
        OUTPUT ${open_world_dir}/cooked/maps/test.world
        COMMAND astral_cook ${open_world_dir}/source -o ${open_world_dir}/cooked -q
        DEPENDS ${open_world_dir}/source/maps/test.world astral_cook
        COMMENT "Cooking the open world test map"
        VERBATIM)
    add_custom_target(open_world_assets ALL DEPENDS ${open_world_dir}/cooked/maps/test.world)
    add_executable(open_world open_world/main.cpp)
    target_link_libraries(open_world PRIVATE Astral::App)
    target_compile_definitions(open_world PRIVATE ASTRAL_OPEN_WORLD_ASSETS="${open_world_dir}/cooked")
    astral_set_warnings(open_world)
    add_dependencies(open_world open_world_assets)
endif()
